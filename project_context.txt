import os

def export_project_to_txt(output_filename="orderping_source_code.txt"):
    ignore_dirs = {'.git', '__pycache__', 'venv', 'env', '.idea', '.vscode'}
    ignore_exts = {'.pyc', '.png', '.jpg', '.jpeg', '.pdf', '.sqlite3'}

    with open(output_filename, 'w', encoding='utf-8') as outfile:
        outfile.write("=== مشروع OrderPing - الكود المصدري ===\n\n")
        
        for root, dirs, files in os.walk('.'):
            dirs[:] = [d for d in dirs if d not in ignore_dirs]

            for file in files:
                ext = os.path.splitext(file)[1].lower()
                if ext in ignore_exts or file == output_filename or file == "export_repo.py":
                    continue

                filepath = os.path.join(root, file)
                outfile.write(f"\n{'='*50}\n")
                outfile.write(f"File Path: {filepath}\n")
                outfile.write(f"{'='*50}\n\n")

                try:
                    with open(filepath, 'r', encoding='utf-8') as infile:
                        outfile.write(infile.read())
                        outfile.write("\n")
                except Exception as e:
                    outfile.write(f"[تعذر قراءة الملف: {e}]\n")

    print(f"✅ تم تجميع الأكواد بنجاح في ملف: {output_filename}")

if __name__ == "__main__":
    export_project_to_txt()

import logging
import re

class PIIRedactionFilter(logging.Filter):
    def filter(self, record):
        # إخفاء أي رقم تليفون (أرقام تبدأ بـ + أو أرقام متتالية طويلة)
        record.msg = re.sub(r'\+?\d{8,15}', '[REDACTED_PHONE]', str(record.msg))
        return True

def setup_logger(name):
    logger = logging.getLogger(name)
    if not logger.handlers:
        handler = logging.StreamHandler()
        formatter = logging.Formatter('[%(asctime)s] %(levelname)s in %(module)s: %(message)s')
        handler.setFormatter(formatter)
        handler.addFilter(PIIRedactionFilter())
        logger.addHandler(handler)
        logger.setLevel(logging.INFO)
    return logger

import os
from flask_sqlalchemy import SQLAlchemy
from datetime import datetime
from cryptography.fernet import Fernet

db = SQLAlchemy()

# إعداد التشفير
def get_cipher():
    key = os.environ.get('ENCRYPTION_KEY')
    if not key:
        raise ValueError("ENCRYPTION_KEY is missing in environment variables")
    return Fernet(key.encode())

def encrypt_data(data: str) -> str:
    if not data: return data
    return get_cipher().encrypt(data.encode()).decode()

def decrypt_data(token: str) -> str:
    if not token: return token
    try:
        return get_cipher().decrypt(token.encode()).decode()
    except Exception:
        return token # Fallback if already decrypted or invalid

class Merchant(db.Model):
    __tablename__ = 'merchants'
    id = db.Column(db.Integer, primary_key=True)
    store_name = db.Column(db.String(100), nullable=False)
    store_url = db.Column(db.String(255), unique=True, nullable=False)
    webhook_secret = db.Column(db.String(100)) # Shopify HMAC Secret
    whatsapp_app_secret = db.Column(db.String(255)) # WhatsApp HMAC Secret
    is_active = db.Column(db.Boolean, default=True)
    created_at = db.Column(db.DateTime, default=datetime.utcnow)
    updated_at = db.Column(db.DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)
    
    orders = db.relationship('Order', backref='merchant', lazy=True)

class Order(db.Model):
    __tablename__ = 'orders'
    id = db.Column(db.Integer, primary_key=True)
    merchant_id = db.Column(db.Integer, db.ForeignKey('merchants.id'), nullable=False)
    shopify_order_id = db.Column(db.String(100), nullable=False)
    _customer_phone = db.Column('customer_phone', db.String(255), nullable=False) # Encrypted
    status = db.Column(db.String(50), default='pending_whatsapp_sent') # State Machine
    created_at = db.Column(db.DateTime, default=datetime.utcnow)
    updated_at = db.Column(db.DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)

    # Property to handle automatic encryption/decryption
    @property
    def customer_phone(self):
        return decrypt_data(self._customer_phone)

    @customer_phone.setter
    def customer_phone(self, value):
        self._customer_phone = encrypt_data(value)

    # Mandatory Merchant Isolation Query Helper
    @classmethod
    def get_by_merchant(cls, merchant_id):
        return cls.query.filter_by(merchant_id=merchant_id)

import os
import hmac
import hashlib
import base64
import json
import re
from flask import Blueprint, request, jsonify
from app.models import db, Order, Merchant
from app.logger_setup import setup_logger
from app.messaging.whatsapp import send_whatsapp_confirmation

logger = setup_logger('orderping.webhooks')
webhooks = Blueprint('webhooks', __name__)

def verify_shopify_hmac(data: bytes, hmac_header: str, secret: str) -> bool:
    if not secret or not hmac_header:
        return False
    digest = hmac.new(secret.encode('utf-8'), data, hashlib.sha256).digest()
    computed_hmac = base64.b64encode(digest).decode('utf-8')
    return hmac.compare_digest(computed_hmac, hmac_header)

def normalize_phone(raw_phone: str) -> str:
    if not raw_phone: return ""
    cleaned = re.sub(r'[^\d+]', '', str(raw_phone).strip())
    if cleaned.startswith('00'): return '+' + cleaned[2:]
    if not cleaned.startswith('+'):
        return '+20' + cleaned[1:] if cleaned.startswith('0') else '+' + cleaned
    return cleaned

# ==========================================
# 1. Normal Orders Webhook (Uses Merchant Secret)
# ==========================================
@webhooks.route('/shopify', methods=['POST'])
def receive_shopify_order():
    shop_domain = request.headers.get('X-Shopify-Shop-Domain')
    if not shop_domain:
        return jsonify({"error": "Missing Shop Domain"}), 400

    merchant = Merchant.query.filter_by(store_url=shop_domain).first()
    if not merchant:
        return jsonify({"error": "Unauthorized Merchant"}), 401

    hmac_header = request.headers.get('X-Shopify-Hmac-Sha256')
    raw_data = request.get_data()
    
    # استخدام مفتاح التاجر
    if not verify_shopify_hmac(raw_data, hmac_header, merchant.webhook_secret):
        logger.error("Shopify HMAC verification failed")
        return jsonify({"error": "Invalid Signature"}), 401

    data = request.get_json(silent=True)
    if not data:
        return jsonify({"error": "Invalid JSON"}), 400

    shopify_order_id = str(data.get('id'))
    raw_phone = data.get('phone') or (data.get('customer') or {}).get('phone')
    customer_phone = normalize_phone(raw_phone)

    if not shopify_order_id or not customer_phone:
        return jsonify({"error": "Missing order_id or phone"}), 400

    existing_order = Order.get_by_merchant(merchant.id).filter_by(shopify_order_id=shopify_order_id).first()
    if existing_order:
        logger.info(f"Idempotency hit: Order {shopify_order_id} already exists.")
        return jsonify({"success": True, "message": "Already processed"}), 200

    try:
        new_order = Order(
            merchant_id=merchant.id,
            shopify_order_id=shopify_order_id,
            customer_phone=customer_phone,
            status='pending_whatsapp_sent'
        )
        db.session.add(new_order)
        db.session.commit()
        
        logger.info(f"Order {shopify_order_id} saved. Status: pending_whatsapp_sent.")

        customer_name = (data.get('customer') or {}).get('first_name', 'عميلنا العزيز')
        total_price = f"{data.get('total_price', '0.00')} {data.get('currency', 'EGP')}"
        send_whatsapp_confirmation(shopify_order_id, customer_name, customer_phone, total_price)

        return jsonify({"success": True}), 200

    except Exception as e:
        db.session.rollback()
        logger.error(f"Database error: {e}")
        return jsonify({"error": "Internal Server Error"}), 500

# ==========================================
# 2. Shopify Mandatory GDPR Webhooks (Uses Global App Secret)
# ==========================================
@webhooks.route('/shopify/gdpr', methods=['POST'])
def shopify_gdpr_webhooks():
    shop_domain = request.headers.get('X-Shopify-Shop-Domain')
    topic = request.headers.get('X-Shopify-Topic')
    hmac_header = request.headers.get('X-Shopify-Hmac-Sha256')
    
    if not shop_domain or not topic:
        return jsonify({"error": "Missing Headers"}), 400

    # استخدام مفتاح التطبيق العام من البيئة
    GLOBAL_APP_SECRET = os.environ.get('SHOPIFY_APP_SECRET', '')
    raw_data = request.get_data()
    
    if not verify_shopify_hmac(raw_data, hmac_header, GLOBAL_APP_SECRET):
        logger.error(f"GDPR Webhook HMAC failed for {shop_domain}")
        return jsonify({"error": "Invalid Signature"}), 401

    merchant = Merchant.query.filter_by(store_url=shop_domain).first()
    if not merchant:
        return jsonify({"success": True}), 200 

    try:
        if topic == 'shop/redact':
            # التاجر مسح التطبيق: نمسح كل بياناته
            Order.query.filter_by(merchant_id=merchant.id).delete()
            db.session.delete(merchant)
            db.session.commit()
            logger.info(f"GDPR: Shop {shop_domain} uninstalled app. All data deleted.")

        return jsonify({"success": True}), 200

    except Exception as e:
        db.session.rollback()
        logger.error(f"GDPR Webhook Error: {e}")
        return jsonify({"error": "Internal Server Error"}), 500

import os
import requests
from app.logger_setup import setup_logger

logger = setup_logger('orderping.messaging')

def send_whatsapp_confirmation(order_id, customer_name, customer_phone, total_price):
    WHATSAPP_TOKEN = os.environ.get('WHATSAPP_TOKEN', '')
    PHONE_NUMBER_ID = os.environ.get('WHATSAPP_PHONE_NUMBER_ID', '')

    if not WHATSAPP_TOKEN or not PHONE_NUMBER_ID:
        # استخدام الـ logger بدل print عشان فلتر الـ PII يشتغل ويخفي الرقم
        logger.info(f"[SIMULATION] WhatsApp Message sent to {customer_phone}")
        logger.info(f"Body: أهلاً {customer_name}، تم استلام طلبك {order_id} بقيمة {total_price}.")
        return True

    url = f"https://graph.facebook.com/v17.0/{PHONE_NUMBER_ID}/messages"
    headers = {
        "Authorization": f"Bearer {WHATSAPP_TOKEN}",
        "Content-Type": "application/json"
    }
    
    payload = {
        "messaging_product": "whatsapp",
        "to": customer_phone.replace("+", ""),
        "type": "interactive",
        "interactive": {
            "type": "button",
            "body": {
                "text": f"أهلاً {customer_name}،\nتم استلام طلبك رقم {order_id} بقيمة {total_price}.\nيرجى تأكيد أو إلغاء الطلب:"
            },
            "action": {
                "buttons": [
                    {"type": "reply", "reply": {"id": f"confirm_{order_id}", "title": "تأكيد الطلب"}},
                    {"type": "reply", "reply": {"id": f"cancel_{order_id}", "title": "إلغاء الطلب"}}
                ]
            }
        }
    }
    
    try:
        response = requests.post(url, headers=headers, json=payload)
        response.raise_for_status()
        logger.info(f"WhatsApp sent successfully to {customer_phone}")
        return True
    except Exception as e:
        logger.error(f"Failed to send WhatsApp to {customer_phone}: {e}")
        return False
